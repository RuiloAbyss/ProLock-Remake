program SmartHome {
	string inputPass = "my_secret_key"

   lock front_door() {
      state:
          boolean is_locked = true
          $PASS = "my_secret_key"
   }

   clock main_clock() {
      state:
         $TIME = <00:01>
         moment lock_time = <22:00>
         moment unlock_time = <07:00>
         moment report_time = <08:00>
   }

   // Rutina consolidada que maneja todas las acciones programadas.
   routine daily_schedule {

      // Accion que bloquea la puerta a la hora definida.
      action night_lock {
         when: (main_clock.$TIME == main_clock.lock_time) -> front_door.force_lock
      }

      // Accion que desbloquea la puerta a la hora definida.
      action morning_unlock {
         when: (main_clock.$TIME == main_clock.unlock_time) -> front_door.force_unlock
      }
       
		action check_pass {
	    	 when: (front_door.$PASS == 	inputPass) -> 
			   front_door.force_unlock,
				front_door.is_locked = true
		}

      //Accion que reporta el estado de la puerta por la manana.
      action report_status {
         when: (main_clock.$TIME.check() == main_clock.report_time.check()) ->
            show("--- Reporte de Estado Diario ---"),
            show("Estado actual de la puerta:"),
            show((front_door).check())
      }
   }
}
