program SmartHome_Test_Suite {
	string global_pass = "admin"
	number version = 1.0

	// --- 1. ERROR LÉXICO: Símbolo inválido ---
	// El símbolo '#' no pertenece al lenguaje ProLock.
	# Compilador v1.0 

   lock front_door() {
      state:
          boolean is_locked = true
          $PASS = "my_secret_key"
   }

   lock back_door() {
      state:
          boolean is_locked = false
   }

   clock main_clock() {
      state:
         $TIME = <00:01>
         moment lock_time = <22:00>
         moment report_time = <08:00>
   }

   routine security_checks {

      // --- 2. ERROR SEMÁNTICO: Variable no declarada ---
      // 'back_door' existe, pero 'back_dor' (con una 'o') no.
      action check_back_door {
         when: (back_dor.$PASS == global_pass) -> back_door.force_lock
      }

      // --- 3. ERROR SINTÁCTICO: Estructura mal formada ---
      // A la cláusula 'when' le falta la flecha '->' antes de la consecuencia.
      action failed_lock {
         when: (main_clock.$TIME == main_clock.lock_time) front_door.force_lock
      }

      // --- 4. ERROR SEMÁNTICO: Tipos incompatibles en comparación ---
      // No se puede comparar un 'string' ($PASS) con un 'moment' (lock_time).
      action invalid_comparison {
         when: (front_door.$PASS == main_clock.lock_time) -> front_door.force_unlock
      }

      // --- 5. ERROR SEMÁNTICO: Asignación de tipo incorrecto ---
      // No se puede asignar un 'string' a la variable 'is_locked' de tipo 'boolean'.
      action bad_assignment {
          when: (front_door.$PASS == global_pass) -> front_door.is_locked = "locked"
      }

      // --- 6. ERROR SEMÁNTICO: Acción no válida para el objeto ---
      // 'force_lock' es una acción para 'lock', no para 'clock'.
      action invalid_action_call {
          when: (main_clock.$TIME == main_clock.report_time) -> main_clock.force_lock
      }
   }

   // --- 7. ADVERTENCIA: Variable no usada ---
   // La variable global 'version' se declara pero nunca se utiliza.
}
